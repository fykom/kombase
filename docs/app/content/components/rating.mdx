## Components Layout

```tsx
import { Rating, RatingItem } from "kombase";

export default function RatingDemo() {
  return (
    <Rating defaultValue={3}>
      <RatingItem />
      <RatingItem />
      <RatingItem />
      <RatingItem />
      <RatingItem />
    </Rating>
  );
}
```
